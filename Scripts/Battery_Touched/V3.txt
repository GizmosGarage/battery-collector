local Players = game:GetService("Players")
local batteryPart = script.Parent
local battery = batteryPart.Parent

batteryPart.Touched:Connect(function(hit)
	local character = hit.Parent
	local player = Players:GetPlayerFromCharacter(character)

	if player then
		print(player.Name .. " collected the battery!")
		battery:Destroy()
	end
end)
