local Players = game:GetService("Players")

local battery = script.Parent
local hitbox = battery.Hitbox

-- .Touched fires MANY times in a fraction of a second (one per body part,
-- plus physics jitter). Without this guard, a single battery could count
-- 3-4 times before Destroy() actually takes effect. Once we've collected,
-- every later touch bails out immediately.
local collected = false

hitbox.Touched:Connect(function(hit)
	if collected then
		return
	end

	local character = hit.Parent
	local player = Players:GetPlayerFromCharacter(character)

	if player then
		collected = true

		-- Walk down to the value PlayerSetup created: player > leaderstats > Batteries.
		-- FindFirstChild returns nil instead of erroring if it's missing, so we
		-- check before using it.
		local leaderstats = player:FindFirstChild("leaderstats")
		if leaderstats then
			local batteries = leaderstats:FindFirstChild("Batteries")
			if batteries then
				batteries.Value += 1
				print(player.Name .. " collected a battery! Total: " .. batteries.Value)
			end
		end

		battery:Destroy()
	end
end)
